       readonly passedAiAuthKeys: readonly string[];
     }

2. In `src/core/process.ts`:
   - Define and export `REVIEWED_RUNTIME_ALLOWLIST = new Set([ ...PLATFORM_FLOOR_ENVIRONMENT, "PATH", "PATHEXT", "COMSPEC", "SHELL", "HOME", "USERPROFILE", "USER", "USERNAME", "TEMP", "TMP", "TMPDIR", "SYSTEMROOT", "SYSTEMDRIVE", "LANG", "LC_ALL", "LC_CTYPE", "TZ", "TERM", "TERM_PROGRAM", "COLORTERM", "EDITOR", "VISUAL", "NODE_EXTRA_CA_CERTS", "SSL_CERT_FILE", "SSL_CERT_DIR" ])`.
   - Define and export `AI_AUTH_PREFIXES = ["OPENAI_", "ANTHROPIC_", "GEMINI_", "GOOGLE_", "AZURE_OPENAI_", "COPILOT_", "GH_COPILOT_", "GITHUB_", "OPENROUTER_", "TOGETHER_", "GROQ_", "MISTRAL_", "DEEPSEEK_"]`.
   - Implement `scrubEnvironmentForOffTree(sourceEnv: NodeJS.ProcessEnv = process.env, injectedEnv: Record<string, string> = {}): ScrubbedEnvironmentResult`:
     - Initialize `cleanEnv = {}`, `scrubbedKeys = []`, `passedRuntimeKeys = []`, `passedAiAuthKeys = []`.
     - Iterate `[key, value]` of `sourceEnv`:
       - If `value === undefined`, continue.
       - Let `upper = key.toUpperCase()`.
       - If `upper.startsWith("ALPHA_AOS_") && upper !== "ALPHA_AOS_DEFAULT_MODE"`:
         - Push to `scrubbedKeys` (suppress internal management variables).
         - Continue.
       - If `REVIEWED_RUNTIME_ALLOWLIST.has(upper)`:
         - `cleanEnv[key] = value`.
         - `passedRuntimeKeys.push(key)`.
         - Continue.
       - If `AI_AUTH_PREFIXES.some(prefix => upper.startsWith(prefix))`:
         - `cleanEnv[key] = value`.
         - `passedAiAuthKeys.push(key)`.
         - Continue.
       - Else:
         - Push to `scrubbedKeys` (scrub ambient database, cloud credentials, tokens, sentinels).
     - Inject harness-specific environment variables (`injectedEnv`):
       - `cleanEnv[k] = v`.
     - Return `{ env: cleanEnv, scrubbedKeys, passedRuntimeKeys, passedAiAuthKeys }`.

3. In `src/shim-dispatch.ts`:
   - In `off` mode execution branch, apply `scrubEnvironmentForOffTree(process.env, exclusion.env)` to child process spawn options, guaranteeing that ambient secrets and sentinels are completely scrubbed before the native harness boots.
